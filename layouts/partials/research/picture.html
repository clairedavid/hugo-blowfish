{{- /* Resolve one research image by slug + filename stem and emit the right markup for whatever format(s) scripts/build-research-data.py actually copied for it: SVG alone (the two PDF plots), a WebP+raster pair (the GAE diagrams, the PINN card), a WebP file with no raster sibling (the ATLAS map, a PNG over 500KB converted to lossless WebP with nothing else kept, see build-research-data.py's module docstring), or a single raster file (the portraits). Params: .slug .stem .class .alt .title (.loading defaults to "lazy") An image with a sibling -still.webp (an animated GIF's last frame) also gets a pointing at the still, so visitors who ask for less motion see it instead of the animation. Images without a sibling are unaffected. Looking the file up here, at render time, rather than trusting the data file to say which format a stem has, means a missing asset fails the build with errorf instead of becoming a broken in production. */ -}} {{- $slug := .slug -}} {{- $stem := .stem -}} {{- $class := .class | default "" -}} {{- $alt := .alt | default "" -}} {{- $title := .title | default "" -}} {{- $loading := .loading | default "lazy" -}} {{- $svg := resources.Get (printf "research/%s/%s.svg" $slug $stem) -}} {{- if $svg -}} {{- else -}} {{- $webp := resources.Get (printf "research/%s/%s.webp" $slug $stem) -}} {{- $raster := or (resources.Get (printf "research/%s/%s.png" $slug $stem)) (resources.Get (printf "research/%s/%s.jpg" $slug $stem)) (resources.Get (printf "research/%s/%s.gif" $slug $stem)) -}} {{- $still := resources.Get (printf "research/%s/%s-still.webp" $slug $stem) -}} {{- if and (not $raster) (not $webp) -}} {{- errorf "research/picture: no asset found for %s/%s (looked for .svg, .webp+.png, .webp+.jpg, .webp, .png, .jpg)" $slug $stem -}} {{- end -}} {{- if and $webp $raster -}} {{- with $still }} {{- end }} {{- else -}} {{- $only := $raster | default $webp -}} {{- with $still }} {{ end }} {{- if $still }}{{ end }} {{- end -}} {{- end -}}
